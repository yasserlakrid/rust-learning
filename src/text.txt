pub fn insert_rc(&mut self , value : i32 ){
        let new_node = Node {
            rc : None , lc : None , value : value 
        };

        match &mut self.root {
            Some(root) => {
               root.rc = Some(Box::new(new_node)) ;
            },
            None => self.root =Some(Box::new(new_node))  ,
        }
    }
